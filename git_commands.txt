1. git clone - клонирование репозитория.
2. git status - просмотр статуса файлов.
3. git add - добавление файлов в индекс.
4. git commit - фиксация изменений.
5. git push - отправка изменений на сервер.
6. git pull - получение обновлений с сервера.
7. git init - создание репозитория.
8. git log - просмотр истории коммитов.
9. git branch - управление ветками.
10. git checkout - переключение веток.
